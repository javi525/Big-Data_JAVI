# 3. Precio de un producto y cálculo del IVA (21%)

iva = 0.21
precio_total = precio_producto * (1 + iva)
print(f"Precio total con IVA: {precio_total}")

# 5. Determinar si un número es positivo, negativo o cero

if numero > 0:
    print("El número es positivo.")
elif numero < 0:
    print("El número es negativo.")
else:
    print("El número es cero.")

# 14. Área de un círculo (A = pi * r^2)
import math
radio = 4.0
area = math.pi * (radio ** 2)
print(f"Área del círculo: {area:.2f}")

# 15. Promedio de tres notas e indicar si está aprobado (>= 5)
n1, n2, n3 = 6.0, 4.5, 7.0
promedio = (n1 + n2 + n3) / 3
estado = "Aprobado" if promedio >= 5 else "Reprobado"
print(f"Promedio: {promedio:.2f} ({estado})")

# 16. Números pares entre 1 y 30

# 17. Convertir Celsius a Fahrenheit

# 18. Convertir minutos a horas y minutos
